import copy
import json
from pathlib import Path
import unittest
from unittest.mock import patch

from capacity_experiment import order, decide
from screen_q8_steady import KIND, CAPTURES, configs, observations, operators, revalidate

ROOT = Path(__file__).resolve().parents[2]


class Q8SteadyTest(unittest.TestCase):
    def setUp(self):
        self.base = json.loads((ROOT/'docs/benchmarks/2026-09-10-slru-screen/raw/pair-0-clock.json').read_text())
        state = self.base['runs'][0]['before']; metal = state['metal']
        self.identity = dict(build=metal['build_fingerprint'],artifact_revision=state['artifact_revision'],
            prepared_manifest_sha256=state['prepared']['manifest_sha256'],device=metal['device'],
            physical_bytes=metal['physical_bytes'],budget_bytes=12*1024**3)

    def raw(self, name):
        raw = copy.deepcopy(self.base); config = next(c for c in configs() if c['name']==name)
        raw.update(gpu_reference_mode='off',gpu_references=[])
        for row in raw['runs']:
            row['gpu_reference_mode']='off'
            for state in [row['before'],row['after'],*[v[k] for v in row['phases'].values() for k in ('before','after')]]:
                state['metal']['kernels'].update(policy=config['kernel_policy'],q8_decode_rows=config['q8_decode_rows'])
                if name=='candidate': state['metal']['kernel_dispatches']['q8_mv_packed_r2_w8']=100
        return raw

    def operator(self, layer=0, stage='gdn', repetitions=5):
        raw = dict(kind='captured_q8_decode_screen',build_fingerprint=self.identity['build'],
            artifact_revision=self.identity['artifact_revision'],exact=True,measurements=[])
        manifest = dict(kind='captured_affine_operators',build_fingerprint=self.identity['build'],
            artifact_revision=self.identity['artifact_revision'],cases=[])
        for n in (512,1024,2560):
            matrix=dict(K=2560,N=n,rows=1,bits=8,fused=False,gathered=False,group=64)
            key=str(n); manifest['cases'].append(dict(matrix=matrix,phase='decode',
                context=dict(layer=layer,stage=stage),tensors=dict(x=dict(sha256=key))))
            for rep in range(repetitions):
                for rows in ((0,2) if rep%2==0 else (2,0)):
                    raw['measurements'].append(dict(matrix=matrix,case=key,repetition=rep,q8_decode_rows=rows,
                        tile=1,output_rows=2,gate_pair=False,wall_ns=100 if rows else 200,exact=True))
        raw['source']=manifest
        return raw

    def test_only_q8_changes_and_memory_must_match(self):
        a,b=configs()
        self.assertEqual({k for k in a if a[k]!=b[k]}, {'name','kernel_policy','q8_decode_rows'})
        expected={}
        for config in configs(): observations(self.raw(config['name']),self.identity,config,self.base['workloads'],expected)
        for change in (lambda r:r['runs'][0]['before']['memory_plan'].update(expert_slots=1460),
                       lambda r:r['runs'][1]['before'].update(short_append_tokens=128),
                       lambda r:r.update(gpu_reference_mode='resident-q8-v1'),
                       lambda r:r['runs'][0]['after']['metal']['kernels'].update(profile=True),
                       lambda r:r['runs'][0]['after']['metal']['kernels'].update(affine_rows=2),
                       lambda r:r['runs'][0]['after']['metal']['kernel_dispatches'].pop('q8_mv_packed_r2_w8'),
                       lambda r:r['runs'][1].update(reused_tokens=105)):
            raw=self.raw('candidate');change(raw)
            with self.assertRaises(ValueError):observations(raw,self.identity,b,self.base['workloads'],expected)

    def test_operator_identity_pairing_and_exactness(self):
        self.assertEqual(len(operators(self.operator(),self.identity,0,'gdn')),3)
        self.assertEqual(len(operators(self.operator(repetitions=20),self.identity,0,'gdn',20)),3)
        with self.assertRaises(ValueError):operators(self.operator(),self.identity,0,'gdn',20)
        with self.assertRaises(ValueError):operators(self.operator(),self.identity,0,'gdn',10)
        for change in (lambda r:r.update(build_fingerprint='stale'),
                       lambda r:r['source']['cases'].pop(),
                       lambda r:r['source']['cases'][0].update(phase='prefill'),
                       lambda r:r['measurements'].reverse(),
                       lambda r:r['measurements'][0].update(output_rows=1),
                       lambda r:r['measurements'][0].update(exact=False),
                       lambda r:r['measurements'][0].update(wall_ns=float('nan'))):
            raw=self.operator();change(raw)
            with self.assertRaises(ValueError):operators(raw,self.identity,0,'gdn')

    def test_summary_reconstructs_originals_and_rejects_edits(self):
        originals={}; measurements=[]; expected={}; captures=[]
        for layer,stage in CAPTURES:
            key=f'operator-{layer}';raw=self.operator(layer,stage);originals[key]=raw
            captures.append(dict(sha256=key,layer=layer,stage=stage,cases=operators(raw,self.identity,layer,stage)))
        for pair,name in order(2):
            key=f'{pair}-{name}';raw=self.raw(name);originals[key]=raw
            config=next(c for c in configs() if c['name']==name)
            measurements.append(dict(pair=pair,configuration=name,sha256=key,
                requests=observations(raw,self.identity,config,self.base['workloads'],expected)))
        summary=dict(kind=KIND,complete=True,configurations=configs(),gpu_reference_mode='off',metal_validation=False,
            identity=self.identity,correctness_sources=[],correctness={'exact':True},operator_sources=captures,operator_repetitions=5,
            measurements=measurements,workload=self.base['workloads'],**decide(measurements,2))
        with patch('screen_q8_steady.correctness',return_value={'exact':True}):
            self.assertFalse(revalidate(summary,originals.__getitem__)['advance_to_confirmation'])
            for change in (lambda r:r.update(complete=False),lambda r:r['measurements'].pop(),
                           lambda r:r['measurements'][1].update(sha256=r['measurements'][0]['sha256']),
                           lambda r:r['measurements'][0]['requests'][0].update(request_ms=1),
                           lambda r:r.update(advance_to_confirmation=True),lambda r:r['operator_sources'].pop()):
                altered=copy.deepcopy(summary);change(altered)
                with self.assertRaises(ValueError):revalidate(altered,originals.__getitem__)


if __name__=='__main__': unittest.main()
